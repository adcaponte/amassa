export async function proximoNumeroDeOrcamento(
  tx: TransacaoDoBanco,
  ano: number,
): Promise<number> {
  const [contador] = await tx
    .insert(contadoresOrcamento)
    .values({ ano, ultimoNumero: 1 })
    .onConflictDoUpdate({
      target: contadoresOrcamento.ano,
      set: { ultimoNumero: sql`${contadoresOrcamento.ultimoNumero} + 1` },

    .returning({ ultimoNumero: contadoresOrcamento.ultimoNumero });

### Upload e Processamento de Fotos — `lib/orcamentos/fotos.ts`

**Fonte:** RESEARCH.md, Pattern 3 (validação + redimensionamento com `sharp`)

**Função de validação e redimensionamento:**
```typescript
import { fileTypeFromBuffer } from "file-type";
import sharp from "sharp";

const TIPOS_ACEITOS = new Set(["image/jpeg", "image/png", "image/webp", "image/heic", "image/heif"]);
const TAMANHO_MAXIMO_BYTES = 15_000_000;

export async function validarEProcessarFoto(bytes: Buffer): Promise<{ ok: boolean; erro?: string; resultado?: Buffer }> {
  if (bytes.length > TAMANHO_MAXIMO_BYTES) {
    return { ok: false, erro: "O arquivo é muito grande." };
  }

  const tipo = await fileTypeFromBuffer(bytes);
  if (!tipo || !TIPOS_ACEITOS.has(tipo.mime)) {
    return { ok: false, erro: "Esse arquivo não é uma imagem. Escolha uma foto." };
  }

  try {
    const reduzida = await sharp(bytes)
      .rotate()  // auto-orient pelo EXIF, depois remove a tag
      .resize({ width: 1600, height: 1600, fit: "inside", withoutEnlargement: true })
      .jpeg({ quality: 82, mozjpeg: true })
      .toBuffer();

    return { ok: true, resultado: reduzida };
  } catch (erro) {
    return { ok: false, erro: "Não deu para processar a foto." };
  }
}
```

import fs from "fs/promises";

export const runtime = "nodejs";

    });

### Tratamento de Erros — Drizzle e SQLSTATE

**Fonte:** `lib/erro/postgres.ts` (tratamento de `ehViolacaoDeChaveEstrangeira()`)

**Padrão:**

Erros do banco vêm em `erro.cause.code` (não `erro.code`), porque Drizzle embrulha o SQLSTATE (nota de memória do projeto).

```typescript
try {
  await db.insert(orcamentos).values(novoOrcamento);
} catch (erro) {
  if (ehViolacaoDeChaveEstrangeira(erro)) {
    return { ok: false, erro: "Uma das referências não existe mais." };
  }
  throw erro;  // erro inesperado — deixar subir
}
```
